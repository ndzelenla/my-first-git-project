## Step 2
CREATE TABLE customers (
    customer_id INT PRIMARY KEY AUTO_INCREMENT,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    phone VARCHAR(20),
    address VARCHAR(200),
    city VARCHAR(50),
    state VARCHAR(2),
    zip_code VARCHAR(10),
    country VARCHAR(50) DEFAULT 'USA',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

- Represents: Individual or business customers who can place orders

CREATE TABLE orders (
    order_id INT PRIMARY KEY AUTO_INCREMENT,
    customer_id INT NOT NULL,
    order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    order_status VARCHAR(20) DEFAULT 'pending' 
        CHECK (order_status IN ('pending', 'processing', 'shipped', 'delivered', 'cancelled')),
    total_amount DECIMAL(10,2) DEFAULT 0.00 CHECK (total_amount >= 0),
    shipping_address VARCHAR(200),
    shipping_city VARCHAR(50),
    shipping_state VARCHAR(2),
    shipping_zip VARCHAR(10),
    shipping_method VARCHAR(30),
    shipping_cost DECIMAL(6,2) DEFAULT 0.00,
    tax_amount DECIMAL(6,2) DEFAULT 0.00,
    payment_method VARCHAR(30),
    payment_status VARCHAR(20) DEFAULT 'pending'
        CHECK (payment_status IN ('pending', 'authorized', 'paid', 'refunded')),
    notes TEXT,
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id)
);

-- Represents: Complete order Transactions

## 2
-- Primary Key: customer_id
## Why appropriate:
- Guaranteed uniqueness: Database generates sequential IDs
- Stability: Never changes (vs. email/phone which can change)
-- Privacy: Doesn't expose personal data in URLs or logs
- Business independence: Not tied to external systems
- Simplicity: Single-column key, easy to reference

-- Primary Key: order_id
## Why appropriate:
- Sequential generation: Naturally orders by creation time
- Multiple systems: External order IDs (Shopify, eBay) can be stored separately
- Audit trail: Sequential IDs help identify missing/gapped orders

## 3 & 4 together
- Relationship: One customer → Many orders (1:N)
- Connection: orders.customer_id → customers.customer_id
- Business Logic: Every order must be placed by a valid customer

WITH customer_transactions AS (
    SELECT 
        customer_id,
        order_date,
        total_amount,
        -- Running totals
        SUM(total_amount) OVER (
            PARTITION BY customer_id 
            ORDER BY order_date 
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS running_total,
        -- Moving averages (30-day, 90-day)
        AVG(total_amount) OVER (
            PARTITION BY customer_id 
            ORDER BY order_date 
            RANGE BETWEEN INTERVAL '30 days' PRECEDING AND CURRENT ROW
        ) AS avg_30day,
        -- Percent of total per customer
        total_amount * 100.0 / SUM(total_amount) OVER (
            PARTITION BY customer_id
        ) AS pct_of_customer_total,
        -- Sequence patterns (days between orders)
        LAG(order_date) OVER (
            PARTITION BY customer_id 
            ORDER BY order_date
        ) AS previous_order_date,
        EXTRACT(DAY FROM order_date - LAG(order_date) OVER (
            PARTITION BY customer_id 
            ORDER BY order_date
        )) AS days_since_last_order,
        -- Order rank within customer
        ROW_NUMBER() OVER (
            PARTITION BY customer_id 
            ORDER BY order_date
        ) AS order_sequence,
        -- Monetary decile across all customers
        NTILE(10) OVER (
            ORDER BY total_amount DESC
        ) AS amount_decile
    FROM orders
    WHERE order_date >= '2023-01-01'
)
SELECT 
    customer_id,
    order_date,
    total_amount,
    running_total,
    avg_30day,
    ROUND(pct_of_customer_total, 2) AS pct_of_customer_total,
    days_since_last_order,
    order_sequence,
    amount_decile,
    -- Customer lifecycle stage based on sequence
    CASE 
        WHEN order_sequence = 1 THEN 'First-time Buyer'
        WHEN order_sequence BETWEEN 2 AND 5 THEN 'Early-stage'
        WHEN order_sequence BETWEEN 6 AND 20 THEN 'Growth-stage'
        WHEN order_sequence > 20 THEN 'Mature'
    END AS customer_lifecycle_stage,
    -- Purchase frequency category
    CASE 
        WHEN days_since_last_order IS NULL THEN 'New'
        WHEN days_since_last_order <= 7 THEN 'Hyper-active'
        WHEN days_since_last_order <= 30 THEN 'Active'
        WHEN days_since_last_order <= 90 THEN 'Dormant'
        ELSE 'At-risk'
    END AS engagement_level
FROM customer_transactions
ORDER BY customer_id, order_date;
