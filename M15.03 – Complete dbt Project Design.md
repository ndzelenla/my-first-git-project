## stg_users.sql
-- models/staging/stg_users.sql
-- Cleans raw users: trims names, lowercases email, uppercases country,
-- casts signup_date to DATE, filters out legacy 'trial' plan users

SELECT
    user_id,
    TRIM(name)                  AS user_name,
    LOWER(TRIM(email))           AS email,
    plan                          AS user_plan,
    CAST(signup_date AS DATE)     AS signup_date,
    UPPER(TRIM(country))          AS country_code

FROM {{ source('raw', 'raw_users') }}

WHERE plan NOT IN ('trial')  -- Exclude legacy trial users
  AND email IS NOT NULL

## stg_subscriptions.sql
-- models/staging/stg_subscriptions.sql
-- Cleans raw subscriptions: casts dates, filters out $0 MRR,
-- derives is_active flag, renames columns for clarity

SELECT
    subscription_id,
    user_id,
    plan                          AS subscription_plan,
    CAST(start_date AS DATE)      AS subscription_start_date,
    CAST(end_date AS DATE)        AS subscription_end_date,
    mrr                           AS monthly_recurring_revenue,
    CASE 
        WHEN end_date IS NULL THEN TRUE
        ELSE FALSE
    END                           AS is_active

FROM {{ source('raw', 'raw_subscriptions') }}

WHERE mrr > 0  -- Exclude free tier subscriptions ($0 MRR)
  AND start_date IS NOT NULL

## stg_events.sql
-- models/staging/stg_events.sql
-- Cleans raw events: casts event_date to DATE, filters out 'test' events,
-- excludes properties_json to keep model lightweight

SELECT
    event_id,
    user_id,
    event_type,
    CAST(event_date AS DATE)      AS event_date

FROM {{ source('raw', 'raw_events') }}

WHERE event_type != 'test'  -- Exclude internal QA test events
  AND user_id IS NOT NULL

## stg_invoices.sql
-- models/staging/stg_invoices.sql
-- Cleans raw invoices: casts invoice_date to DATE, filters out void invoices,
-- renames status to invoice_status

SELECT
    invoice_id,
    user_id,
    amount                        AS invoice_amount,
    CAST(invoice_date AS DATE)    AS invoice_date,
    status                        AS invoice_status

FROM {{ source('raw', 'raw_invoices') }}

WHERE status != 'void'  -- Exclude voided invoices
  AND amount >= 0       -- Include $0 invoices (free tier) but not negative

## stg_support_tickets.sql

-- models/staging/stg_support_tickets.sql
-- Cleans raw support tickets: casts timestamps to DATE, filters out duplicate tickets,
-- computes resolution hours for resolved tickets

SELECT
    ticket_id,
    user_id,
    subject,
    priority,
    CAST(created_at AS DATE)      AS ticket_created_date,
    CAST(resolved_at AS DATE)      AS ticket_resolved_date,
    status                        AS ticket_status,
    CASE
        WHEN resolved_at IS NOT NULL THEN 
            EXTRACT(EPOCH FROM (resolved_at - created_at)) / 3600
        ELSE NULL
    END                           AS resolution_hours

FROM {{ source('raw', 'raw_support_tickets') }}

WHERE status != 'duplicate'  -- Exclude duplicate tickets
  AND priority IN ('low', 'medium', 'high', 'critical')


## Step 3: Intermediate Models int_users_enriched.sql

-- models/intermediate/int_users_enriched.sql
-- Enriches user data with subscription and activity metrics
-- One row per user with latest subscription and engagement summary

WITH user_base AS (
    SELECT
        user_id,
        user_name,
        email,
        user_plan,
        signup_date,
        country_code
    FROM {{ ref('stg_users') }}
),

latest_subscription AS (
    SELECT DISTINCT ON (user_id)
        user_id,
        subscription_id,
        subscription_plan,
        subscription_start_date,
        subscription_end_date,
        monthly_recurring_revenue,
        is_active
    FROM {{ ref('stg_subscriptions') }}
    ORDER BY user_id, subscription_start_date DESC
),

user_activity AS (
    SELECT
        user_id,
        COUNT(*) AS total_events,
        COUNT(DISTINCT event_type) AS unique_event_types,
        MAX(event_date) AS last_activity_date
    FROM {{ ref('stg_events') }}
    GROUP BY user_id
),

user_support AS (
    SELECT
        user_id,
        COUNT(*) AS total_tickets,
        COUNT(CASE WHEN ticket_status IN ('open', 'in_progress') THEN 1 END) AS open_tickets,
        AVG(resolution_hours) AS avg_resolution_hours
    FROM {{ ref('stg_support_tickets') }}
    GROUP BY user_id
)

SELECT
    -- User fields
    ub.user_id,
    ub.user_name,
    ub.email,
    ub.user_plan,
    ub.signup_date,
    ub.country_code,
    
    -- Subscription fields
    ls.subscription_id,
    ls.subscription_plan,
    ls.subscription_start_date,
    ls.subscription_end_date,
    ls.monthly_recurring_revenue,
    COALESCE(ls.is_active, FALSE) AS has_active_subscription,
    
    -- Activity fields
    COALESCE(ua.total_events, 0) AS total_events,
    COALESCE(ua.unique_event_types, 0) AS unique_event_types,
    ua.last_activity_date,
    EXTRACT(DAY FROM (CURRENT_DATE - ub.signup_date)) AS days_since_signup,
    EXTRACT(DAY FROM (CURRENT_DATE - ua.last_activity_date)) AS days_since_last_activity,
    
    -- Support fields
    COALESCE(us.total_tickets, 0) AS total_tickets,
    COALESCE(us.open_tickets, 0) AS open_tickets,
    us.avg_resolution_hours

FROM user_base ub
LEFT JOIN latest_subscription ls ON ub.user_id = ls.user_id
LEFT JOIN user_activity ua ON ub.user_id = ua.user_id
LEFT JOIN user_support us ON ub.user_id = us.user_id

## int_support_with_users.sql
-- models/intermediate/int_support_with_users.sql
-- Enriches support tickets with user information for support analysis
-- One row per support ticket with user context

SELECT
    -- Ticket fields
    t.ticket_id,
    t.user_id,
    t.subject,
    t.priority,
    t.ticket_created_date,
    t.ticket_resolved_date,
    t.ticket_status,
    t.resolution_hours,
    
    -- User fields
    u.user_name,
    u.email,
    u.user_plan,
    u.signup_date,
    u.country_code,
    u.has_active_subscription,
    u.monthly_recurring_revenue,
    
    -- Computed fields
    CASE
        WHEN t.ticket_status IN ('open', 'in_progress') THEN
            EXTRACT(DAY FROM (CURRENT_DATE - t.ticket_created_date))
        ELSE NULL
    END AS days_open,
    
    CASE t.priority
        WHEN 'critical' THEN 1
        WHEN 'high' THEN 2
        WHEN 'medium' THEN 3
        WHEN 'low' THEN 4
    END AS priority_rank,
    
    CASE
        WHEN t.ticket_status IN ('resolved', 'closed') THEN TRUE
        ELSE FALSE
    END AS is_resolved

FROM {{ ref('stg_support_tickets') }} t
LEFT JOIN {{ ref('int_users_enriched') }} u ON t.user_id = u.user_id

## mart_user_health.sql
-- models/marts/mart_user_health.sql
-- User health and churn risk dashboard for Customer Success Team
-- One row per user with health metrics and risk indicators

WITH user_health_base AS (
    SELECT
        user_id,
        user_name,
        email,
        user_plan,
        country_code,
        signup_date,
        has_active_subscription,
        monthly_recurring_revenue,
        total_events,
        unique_event_types,
        last_activity_date,
        days_since_signup,
        days_since_last_activity,
        total_tickets,
        open_tickets,
        avg_resolution_hours
    FROM {{ ref('int_users_enriched') }}
)

SELECT
    -- User profile
    user_id,
    user_name,
    email,
    user_plan,
    country_code,
    signup_date,
    
    -- Subscription status
    has_active_subscription,
    CASE 
        WHEN has_active_subscription THEN monthly_recurring_revenue
        ELSE 0
    END AS current_mrr,
    
    -- Engagement metrics
    total_events,
    unique_event_types,
    last_activity_date,
    days_since_signup,
    days_since_last_activity,
    
    -- Support metrics
    total_tickets,
    open_tickets,
    avg_resolution_hours,
    
    -- Churn risk calculation
    CASE
        WHEN NOT has_active_subscription THEN 'churned'
        WHEN days_since_last_activity > 30 AND open_tickets > 0 THEN 'high'
        WHEN days_since_last_activity > 30 THEN 'medium'
        WHEN days_since_last_activity > 14 THEN 'low'
        ELSE 'healthy'
    END AS churn_risk,
    
    -- Health score (0-100)
    GREATEST(0, LEAST(100,
        CASE WHEN has_active_subscription THEN 40 ELSE 0 END +
        CASE WHEN days_since_last_activity < 7 THEN 30 
             WHEN days_since_last_activity < 14 THEN 20
             WHEN days_since_last_activity < 30 THEN 10 ELSE 0 END +
        CASE WHEN open_tickets = 0 THEN 20 
             WHEN open_tickets = 1 THEN 10 ELSE 0 END +
        CASE WHEN total_events > 10 THEN 10 
             WHEN total_events > 5 THEN 5 ELSE 0 END
    )) AS health_score,
    
    -- Action flags
    CASE WHEN days_since_last_activity > 7 THEN TRUE ELSE FALSE END AS needs_engagement,
    CASE WHEN open_tickets > 0 THEN TRUE ELSE FALSE END AS needs_support_attention

FROM user_health_base

WHERE user_id IS NOT NULL

## mart_revenue_metrics.sql
-- models/marts/mart_revenue_metrics.sql
-- Revenue analytics dashboard for Finance Team and Executive Dashboard
-- Aggregated by month and plan with MRR and invoice metrics

WITH monthly_subscriptions AS (
    SELECT
        DATE_TRUNC('month', subscription_start_date) AS month,
        subscription_plan,
        COUNT(DISTINCT user_id) AS active_subscribers,
        SUM(monthly_recurring_revenue) AS total_mrr
    FROM {{ ref('stg_subscriptions') }}
    WHERE is_active = TRUE
    GROUP BY DATE_TRUNC('month', subscription_start_date), subscription_plan
),

monthly_invoices AS (
    SELECT
        DATE_TRUNC('month', invoice_date) AS month,
        COUNT(*) AS total_invoices,
        SUM(invoice_amount) AS total_invoiced,
        SUM(CASE WHEN invoice_status = 'paid' THEN invoice_amount ELSE 0 END) AS total_collected,
        SUM(CASE WHEN invoice_status = 'overdue' THEN invoice_amount ELSE 0 END) AS total_overdue,
        SUM(CASE WHEN invoice_status = 'unpaid' THEN invoice_amount ELSE 0 END) AS total_unpaid
    FROM {{ ref('stg_invoices') }}
    GROUP BY DATE_TRUNC('month', invoice_date)
),

plan_breakdown AS (
    SELECT
        month,
        subscription_plan,
        active_subscribers,
        total_mrr,
        ROUND(AVG(total_mrr / NULLIF(active_subscribers, 0)), 2) AS avg_revenue_per_user
    FROM monthly_subscriptions
    GROUP BY month, subscription_plan, active_subscribers, total_mrr
)

SELECT
    COALESCE(pb.month, mi.month) AS revenue_month,
    
    -- Plan-level metrics
    pb.subscription_plan,
    pb.active_subscribers,
    pb.total_mrr,
    pb.avg_revenue_per_user,
    
    -- Invoice metrics
    mi.total_invoices,
    mi.total_invoiced,
    mi.total_collected,
    mi.total_overdue,
    mi.total_unpaid,
    ROUND((mi.total_collected / NULLIF(mi.total_invoiced, 0)) * 100, 2) AS collection_rate,
    
    -- Computed metrics
    CASE 
        WHEN pb.total_mrr > 0 THEN 'growing'
        WHEN pb.total_mrr < 0 THEN 'declining'
        ELSE 'stable'
    END AS revenue_trend

FROM plan_breakdown pb
FULL OUTER JOIN monthly_invoices mi ON pb.month = mi.month

ORDER BY revenue_month DESC, pb.subscription_plan

## mart_support_performance.sql
-- models/marts/mart_support_performance.sql
-- Support performance dashboard for Support Team Lead
-- Aggregated by priority, status, and user segments

WITH ticket_metrics AS (
    SELECT
        priority,
        ticket_status,
        user_plan,
        country_code,
        COUNT(*) AS ticket_count,
        AVG(resolution_hours) AS avg_resolution_hours,
        MIN(resolution_hours) AS min_resolution_hours,
        MAX(resolution_hours) AS max_resolution_hours,
        AVG(days_open) AS avg_days_open,
        COUNT(CASE WHEN is_resolved THEN 1 END) AS resolved_count,
        COUNT(CASE WHEN NOT is_resolved THEN 1 END) AS open_count
    FROM {{ ref('int_support_with_users') }}
    GROUP BY priority, ticket_status, user_plan, country_code
),

priority_summary AS (
    SELECT
        priority,
        COUNT(*) AS total_tickets_by_priority,
        AVG(resolution_hours) AS avg_resolution_by_priority,
        SUM(CASE WHEN ticket_status IN ('open', 'in_progress') THEN 1 ELSE 0 END) AS open_by_priority
    FROM {{ ref('int_support_with_users') }}
    GROUP BY priority
),

plan_summary AS (
    SELECT
        user_plan,
        COUNT(*) AS total_tickets_by_plan,
        AVG(resolution_hours) AS avg_resolution_by_plan,
        COUNT(DISTINCT user_id) AS unique_users_with_tickets
    FROM {{ ref('int_support_with_users') }}
    GROUP BY user_plan
)

SELECT
    -- Priority breakdown
    tm.priority,
    ps.total_tickets_by_priority,
    ps.open_by_priority,
    ps.avg_resolution_by_priority,
    
    -- Status breakdown
    tm.ticket_status,
    tm.ticket_count,
    tm.avg_resolution_hours,
    tm.min_resolution_hours,
    tm.max_resolution_hours,
    tm.avg_days_open,
    tm.resolved_count,
    tm.open_count,
    
    -- User segment breakdown
    tm.user_plan,
    tm.country_code,
    ps_plan.total_tickets_by_plan,
    ps_plan.avg_resolution_by_plan,
    ps_plan.unique_users_with_tickets,
    
    -- SLA compliance
    CASE
        WHEN tm.priority = 'critical' AND tm.avg_resolution_hours <= 4 THEN 'SLA Met'
        WHEN tm.priority = 'high' AND tm.avg_resolution_hours <= 24 THEN 'SLA Met'
        WHEN tm.priority = 'medium' AND tm.avg_resolution_hours <= 72 THEN 'SLA Met'
        WHEN tm.priority = 'low' AND tm.avg_resolution_hours <= 168 THEN 'SLA Met'
        ELSE 'SLA Breached'
    END AS sla_status

FROM ticket_metrics tm
LEFT JOIN priority_summary ps ON tm.priority = ps.priority
LEFT JOIN plan_summary ps_plan ON tm.user_plan = ps_plan.user_plan

WHERE tm.priority IS NOT NULL
ORDER BY 
    CASE tm.priority
        WHEN 'critical' THEN 1
        WHEN 'high' THEN 2
        WHEN 'medium' THEN 3
        WHEN 'low' THEN 4
    END
