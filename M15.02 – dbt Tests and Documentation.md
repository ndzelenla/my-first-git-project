## Step 3c: Custom Tests
## Test 1: Revenue must be positive
-- tests/assert_revenue_by_customer_is_positive.sql
-- This test FAILS if any customer has zero or negative total_revenue.
-- dbt convention: the SELECT returns rows that VIOLATE the rule.

SELECT
    customer_id,
    customer_name,
    total_revenue

FROM {{ ref('mart_revenue_by_customer') }}

WHERE total_revenue <= 0

## Test 2: Total orders must be positive

-- tests/assert_customer_has_at_least_one_order.sql
-- This test FAILS if any customer has zero orders in the mart.

SELECT
    customer_id,
    customer_name,
    total_orders

FROM {{ ref('mart_revenue_by_customer') }}

WHERE total_orders <= 0

## Test 3: Revenue per unit should be reasonable
-- tests/assert_product_revenue_per_unit_is_reasonable.sql
-- This test FAILS if any product has revenue_per_unit > 10× its unit_price.
-- A 10× multiplier accounts for variable pricing/discounts but catches extreme outliers.

SELECT
    product_id,
    product_name,
    unit_price,
    revenue_per_unit,
    (revenue_per_unit / NULLIF(unit_price, 0)) AS price_ratio

FROM {{ ref('mart_revenue_by_product') }}

WHERE revenue_per_unit > (unit_price * 10)

## Test 4: First order date must be before last order date
-- tests/assert_first_order_before_last_order.sql
-- This test FAILS if any customer's first_order_date is after their last_order_date.

SELECT
    customer_id,
    customer_name,
    first_order_date,
    last_order_date

FROM {{ ref('mart_revenue_by_customer') }}

WHERE first_order_date > last_order_date

## Test 5: Units sold must be positive
-- tests/assert_product_has_units_sold.sql
-- This test FAILS if any product in the mart has zero or negative units sold.

SELECT
    product_id,
    product_name,
    total_units_sold

FROM {{ ref('mart_revenue_by_product') }}

WHERE total_units_sold <= 0

## Step 5a: Data Contract for mart_revenue_by_customer

# data_contracts/contract_mart_revenue_by_customer.yml

# Data Contract: mart_revenue_by_customer
# Version: 1.0
# Owner: Analytics Engineering Team
# Consumers: Marketing Team, Customer Success, Executive Dashboards

contract:
  model: mart_revenue_by_customer
  description: >
    Revenue metrics aggregated by customer. This is the authoritative source
    for customer-level revenue data. All downstream dashboards and reports
    MUST use this model instead of querying raw or staging tables directly.

  columns:
    - name: customer_id
      type: INTEGER
      nullable: false
      unique: true
      description: "Primary key — unique customer identifier"

    - name: customer_name
      type: VARCHAR
      nullable: false
      unique: false
      description: "Customer full name (trimmed)"

    - name: email
      type: VARCHAR
      nullable: false
      unique: true
      description: "Customer email address (lowercase)"

    - name: country_code
      type: VARCHAR(2)
      nullable: false
      unique: false
      description: "Two-letter country code (uppercase)"

    - name: total_orders
      type: INTEGER
      nullable: false
      unique: false
      constraints:
        - "total_orders > 0"
      description: "Total number of distinct orders"

    - name: total_units_purchased
      type: INTEGER
      nullable: false
      unique: false
      constraints:
        - "total_units_purchased > 0"
      description: "Total product units purchased"

    - name: total_revenue
      type: NUMERIC(12, 2)
      nullable: false
      unique: false
      constraints:
        - "total_revenue > 0"
      description: "Total revenue from completed payments (USD)"

    - name: avg_order_value
      type: NUMERIC(10, 2)
      nullable: false
      unique: false
      constraints:
        - "avg_order_value > 0"
      description: "Average revenue per order"

    - name: first_order_date
      type: DATE
      nullable: false
      unique: false
      constraints:
        - "first_order_date <= last_order_date"
      description: "Date of the customer's earliest order"

    - name: last_order_date
      type: DATE
      nullable: false
      unique: false
      constraints:
        - "last_order_date >= first_order_date"
      description: "Date of the customer's most recent order"

  freshness:
    warn_after: "24 hours"
    error_after: "48 hours"
    description: "Model is refreshed daily. Stale data beyond 48 hours is a critical issue."

  sla:
    availability: "99.5%"
    refresh_schedule: "Daily by 6:00 AM UTC"
    description: "Must be available for the 7:00 AM executive dashboard refresh"
