## Task 3: Implement Column-Level Security
USE ROLE PLATFORM_ADMIN;
USE DATABASE STREAMPULSE_PROD;
USE SCHEMA CURATED;

-- Create masking policies
CREATE OR REPLACE MASKING POLICY email_mask AS (val STRING) 
  RETURNS STRING ->
  CASE
    WHEN CURRENT_ROLE() IN ('DATA_ENGINEER', 'PLATFORM_ADMIN', 'ETL_SERVICE', 'DBT_SERVICE') THEN val
    ELSE '***@' || SPLIT_PART(val, '@', 2)
  END;

CREATE OR REPLACE MASKING POLICY phone_mask AS (val STRING) 
  RETURNS STRING ->
  CASE
    WHEN CURRENT_ROLE() IN ('DATA_ENGINEER', 'PLATFORM_ADMIN', 'ETL_SERVICE', 'DBT_SERVICE') THEN val
    ELSE '***-***-' || RIGHT(val, 4)
  END;

CREATE OR REPLACE MASKING POLICY revenue_mask AS (val NUMBER) 
  RETURNS NUMBER ->
  CASE
    WHEN CURRENT_ROLE() IN ('FINANCE_ANALYST', 'DATA_ENGINEER', 'PLATFORM_ADMIN') THEN val
    ELSE NULL
  END;

-- Create sample customers table with sensitive data
CREATE OR REPLACE TABLE customers (
  customer_id INT,
  name STRING,
  email STRING,
  phone STRING,
  subscription_tier STRING,
  lifetime_revenue NUMBER
);
