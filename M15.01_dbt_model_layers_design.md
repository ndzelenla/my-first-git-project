## Step 1: Staging Models
# Step 1a: stg_orders.sql

-- models/staging/stg_orders.sql
-- Cleans raw orders: casts order_date to DATE, filters out cancelled orders

SELECT
    order_id,
    customer_id,
    product_id,
    CAST(order_date AS DATE)    AS order_date,
    quantity,
    status                      AS order_status

FROM {{ source('raw', 'raw_orders') }}

WHERE status != 'cancelled'

## Step 1b: stg_customers.sql
-- models/staging/stg_customers.sql
-- Cleans raw customers: trims names, validates email contains '@'

SELECT
    customer_id,
    TRIM(name)                  AS customer_name,
    LOWER(TRIM(email))          AS email,
    UPPER(country)              AS country_code,
    CAST(created_at AS DATE)    AS customer_created_date

FROM {{ source('raw', 'raw_customers') }}

WHERE email LIKE '%@%.%'

## Step 1c: stg_products.sql
-- models/staging/stg_products.sql
-- Cleans raw products: standardizes category to lowercase

SELECT
    product_id,
    product_name,
    LOWER(TRIM(category))       AS product_category,
    unit_price

FROM {{ source('raw', 'raw_products') }}

WHERE unit_price > 0


## Step 1d: stg_payments.sql
-- models/staging/stg_payments.sql
-- Cleans raw payments: keeps only completed payments

SELECT
    payment_id,
    order_id,
    amount                      AS payment_amount,
    payment_method,
    CAST(payment_date AS DATE)  AS payment_date,
    status                      AS payment_status

FROM {{ source('raw', 'raw_payments') }}

WHERE status = 'completed'

### Step 2: Intermediate Model
## Step 2a: int_orders_enriched.sql

-- models/intermediate/int_orders_enriched.sql
-- Joins orders with customers, products, and payments into one enriched table

SELECT
    -- Order fields
    o.order_id,
    o.order_date,
    o.quantity,
    o.order_status,

    -- Customer fields
    c.customer_id,
    c.customer_name,
    c.email,
    c.country_code,
    c.customer_created_date,

    -- Product fields
    p.product_id,
    p.product_name,
    p.product_category,
    p.unit_price,

    -- Payment fields
    pay.payment_id,
    pay.payment_amount,
    pay.payment_method,
    pay.payment_date,

    -- Computed fields
    (o.quantity * p.unit_price)  AS line_total

FROM {{ ref('stg_orders') }} AS o

LEFT JOIN {{ ref('stg_customers') }} AS c
    ON o.customer_id = c.customer_id

LEFT JOIN {{ ref('stg_products') }} AS p
    ON o.product_id = p.product_id

LEFT JOIN {{ ref('stg_payments') }} AS pay
    ON o.order_id = pay.order_id

### Step 3: Mart Models
## Step 3a: mart_revenue_by_customer.sql

-- models/marts/mart_revenue_by_customer.sql
-- Aggregates revenue, order count, and average order value per customer

SELECT
    customer_id,
    customer_name,
    email,
    country_code,
    customer_created_date,

    COUNT(DISTINCT order_id)            AS total_orders,
    SUM(quantity)                        AS total_units_purchased,
    SUM(payment_amount)                  AS total_revenue,
    ROUND(
        SUM(payment_amount) / NULLIF(COUNT(DISTINCT order_id), 0),
        2
    )                                    AS avg_order_value,
    MIN(order_date)                      AS first_order_date,
    MAX(order_date)                      AS last_order_date

FROM {{ ref('int_orders_enriched') }}

WHERE payment_amount IS NOT NULL

GROUP BY
    customer_id,
    customer_name,
    email,
    country_code,
    customer_created_date

## Step 3b: mart_revenue_by_product.sql
-- models/marts/mart_revenue_by_product.sql
-- Aggregates revenue, units sold, and order count per product and category

SELECT
    product_id,
    product_name,
    product_category,
    unit_price,

    COUNT(DISTINCT order_id)            AS total_orders,
    SUM(quantity)                        AS total_units_sold,
    SUM(payment_amount)                  AS total_revenue,
    ROUND(
        SUM(payment_amount) / NULLIF(SUM(quantity), 0),
        2
    )                                    AS revenue_per_unit,
    COUNT(DISTINCT customer_id)          AS unique_customers

FROM {{ ref('int_orders_enriched') }}

WHERE payment_amount IS NOT NULL

GROUP BY
    product_id,
    product_name,
    product_category,
    unit_price

sources:

    schema: raw_data

      - name: raw_orders
        description: "Raw orders table — one row per order placed"
        columns:
          - name: order_id
            description: "Primary key — unique order identifier"
          - name: customer_id
            description: "Foreign key to raw_customers"
          - name: product_id
            description: "Foreign key to raw_products"
          - name: order_date
            description: "Timestamp when the order was placed"
          - name: quantity
            description: "Number of units ordered"
          - name: status
            description: "Order status: completed, cancelled, or pending"

      - name: raw_customers
        description: "Raw customers table — one row per registered customer"
        columns:
          - name: customer_id
            description: "Primary key — unique customer identifier"
          - name: name
            description: "Customer full name (may have whitespace)"
          - name: email
            description: "Customer email address"
          - name: country
            description: "Two-letter country code"
          - name: created_at
            description: "Timestamp when the customer account was created"

      - name: raw_products
        description: "Raw products table — one row per product in the catalog"
        columns:
          - name: product_id
            description: "Primary key — unique product identifier"
          - name: product_name
            description: "Product display name"
          - name: category
            description: "Product category (inconsistent casing in raw data)"
          - name: unit_price
            description: "Price per unit in USD"

      - name: raw_payments
        description: "Raw payments table — one row per payment attempt"
        columns:
          - name: payment_id
            description: "Primary key — unique payment identifier"
          - name: order_id
            description: "Foreign key to raw_orders"
          - name: amount
            description: "Payment amount in USD"
          - name: payment_method
            description: "Payment method: credit_card, paypal, bank_transfer"
          - name: payment_date
            description: "Timestamp when the payment was processed"

            description: "Payment status: completed, failed, or pending"


## Step 5c: dbt_project.yml
# dbt_project.yml

name: 'cartwave_dbt'
version: '1.0.0'
config-version: 2

profile: 'cartwave'

model-paths: ["models"]
test-paths: ["tests"]

models:
  cartwave_dbt:
    staging:
      +materialized: view        # Staging models are views (lightweight, always fresh)
    intermediate:
      +materialized: view        # Intermediate models are views
    marts:
      +materialized: table       # Mart models are tables (precomputed for performance)
